cmake_minimum_required(VERSION 3.24)
project(stelar_pro_cpp VERSION 0.1.0 LANGUAGES CXX)
option(STELAR_CUDA "Build CUDA backend" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(stelar_core src/core.cpp src/s4.cpp src/s6.cpp)
target_include_directories(stelar_core PUBLIC include)
target_link_libraries(stelar_core PUBLIC Threads::Threads)
target_compile_options(stelar_core PUBLIC $<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang>:-ffp-contract=off>)
if(STELAR_CUDA)
  # nvcc's compiler-identification probe also needs the narrow glibc/CUDA
  # declaration compatibility shim, before target flags can take effect.
  set(CMAKE_CUDA_FLAGS_INIT "${CMAKE_CUDA_FLAGS_INIT} -Xcompiler=-include,${CMAKE_CURRENT_SOURCE_DIR}/cuda/glibc_math_compat.hpp")
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  add_library(stelar_backend cuda/backend.cu)
  target_compile_options(stelar_backend PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-include,${CMAKE_CURRENT_SOURCE_DIR}/cuda/glibc_math_compat.hpp;-lineinfo>)
else()
  add_library(stelar_backend src/cuda_stub.cpp)
endif()
target_include_directories(stelar_backend PUBLIC include)
add_executable(stelar-pro-cpp src/main.cpp)
target_compile_definitions(stelar-pro-cpp PRIVATE STELAR_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(stelar-pro-cpp PRIVATE stelar_core stelar_backend)
enable_testing()
add_executable(test_s4 test/test_s4.cpp)
target_link_libraries(test_s4 PRIVATE stelar_core stelar_backend)
add_test(NAME s4_components COMMAND test_s4)
add_executable(test_core test/test_core.cpp)
target_link_libraries(test_core PRIVATE stelar_core stelar_backend)
add_test(NAME core_triplet_oracles COMMAND test_core)
add_executable(test_s6 test/test_s6.cpp)
target_link_libraries(test_s6 PRIVATE stelar_core stelar_backend)
add_test(NAME s6_algorithm_oracles COMMAND test_s6)
# White-box checks must execute even in a Release configuration.
foreach(check_target test_core test_s4 test_s6)
  target_compile_options(${check_target} PRIVATE -UNDEBUG)
endforeach()
if(STELAR_CUDA)
  add_executable(test_cuda test/test_cuda.cpp)
  target_link_libraries(test_cuda PRIVATE stelar_core stelar_backend)
  add_test(NAME cuda_parity COMMAND test_cuda)
  set_tests_properties(cuda_parity PROPERTIES SKIP_RETURN_CODE 77)
endif()
